\section{Lemmas on the join of two cycles}
\label{sec:lemmas}

For $m \geq 3$ and $x = (x_{1}, \ldots, x_{m}) \in (\mathbb{R}^{N})^{m}$, with
indices modulo $m$, let
\begin{equation}
  \label{eq:areavector}
  \mathcal{A}(x) = \sum_{j=1}^{m} x_{j} \wedge x_{j+1}.
\end{equation}
Thus $\frac12\mathcal{A}(x)$ is the \emph{area vector} of the closed polygon
$x_{1} \cdots x_{m}$; for $N = 2$ its single coordinate is the signed area given
by the shoelace formula.

\begin{definition}
  \label{def:join}
  For $m, l \geq 3$ let $\mathcal{J}_{m,l} = (m + l, \mathcal{F}_{m,l})$ with
  \[
    \mathcal{F}_{m,l} = \bigl\{(i,\ i+1,\ m+j,\ m+j+1) : i \in \mathbb{Z}_{m},\
      j \in \mathbb{Z}_{l}\bigr\},
  \]
  where the first two entries are read modulo $m$ in $\{1, \ldots, m\}$ and the
  last two modulo $l$ in $\{m+1, \ldots, m+l\}$. For
  $p \in (\mathbb{R}^{4})^{m+l}$ write $a_{i} = p_{i}$, $b_{j} = p_{m+j}$, and
  \[
    A = \mathcal{A}(a_{1}, \ldots, a_{m}), \qquad
    B = \mathcal{A}(b_{1}, \ldots, b_{l}).
  \]
\end{definition}

Each facet of $\mathcal{J}_{m,l}$ consists of an edge of the cycle $(1, \ldots,
m)$ and an edge of the cycle $(m+1, \ldots, m+l)$, so as a simplicial complex
$\mathcal{J}_{m,l}$ is the join of these two cycles. By
Lemma~\ref{lem:freesum} it is the boundary complex of the direct sum of two
regular polygons; the polytope itself is this direct sum, not a join of
polytopes.

By \eqref{eq:signed-volume}
\begin{equation}
  \label{eq:Djoin}
  D(p) = \sum_{i \in \mathbb{Z}_{m}}\sum_{j \in \mathbb{Z}_{l}}
    \det(a_{i}, a_{i+1}, b_{j}, b_{j+1}).
\end{equation}

\begin{lemma}
  \label{lem:freesum}
  Let $m, l \geq 3$, let $u_{1}, u_{2}, u_{3}, u_{4}$ be an orthonormal basis of
  $\mathbb{R}^{4}$, and for $i \in \mathbb{Z}_{m}$, $j \in \mathbb{Z}_{l}$ let
  \[
    a_{i} = \cos\tfrac{2\pi i}{m}\,u_{1} + \sin\tfrac{2\pi i}{m}\,u_{2}, \qquad
    b_{j} = \cos\tfrac{2\pi j}{l}\,u_{3} + \sin\tfrac{2\pi j}{l}\,u_{4}.
  \]
  Then $P = \conv\{a_{i}, b_{j}\} \in \mathcal{P}_{4}(m + l)$, and the point
  $p = (a_{1}, \ldots, a_{m}, b_{1}, \ldots, b_{l})$ realises
  $\mathcal{J}_{m,l}$.
\end{lemma}

\begin{proof}
  In the coordinates of the basis $u_{1}, \ldots, u_{4}$, $P$ is the direct sum
  of two regular polygons with $m$ and $l$ vertices and edges, inscribed in unit
  circles centred at the origin. By the cited statement on direct sums, $P$ is
  $4$-dimensional, all $m + l$ points $a_{i}, b_{j}$ are vertices, and $P$ has
  $ml$ facets. All the points are unit vectors, so $P \in \mathcal{P}_{4}(m +
  l)$.

  It remains to show that each of the $ml$ distinct simplices
  $\conv\{a_{i}, a_{i+1}, b_{j}, b_{j+1}\}$ is a facet. Fix $i \in \mathbb{Z}_{m}$
  and $j \in \mathbb{Z}_{l}$. Let
  \[
    \nu_{i} = \cos\tfrac{(2i+1)\pi}{m}\,u_{1} + \sin\tfrac{(2i+1)\pi}{m}\,u_{2},
    \qquad h_{m} = \cos\tfrac{\pi}{m},
  \]
  so $\nu_{i}$ is the unit vector towards the midpoint of the edge from
  $a_{i}$ to $a_{i+1}$, and $h_{m}$ is the distance of that edge from the
  origin. For every $k$,
  \[
    \nu_{i} \cdot a_{k} = \cos\tfrac{(2k-2i-1)\pi}{m} \leq h_{m},
  \]
  with equality if and only if $2k - 2i - 1 \equiv \pm 1 \pmod{2m}$, that is,
  $k \in \{i, i+1\}$; and $\nu_{i} \cdot b_{k} = 0$, since $\nu_{i}$ lies in
  the plane of $u_{1}, u_{2}$ and $b_{k}$ in that of $u_{3}, u_{4}$. Define
  $\mu_{j}$ and $h_{l} = \cos\frac{\pi}{l}$ in the same way for the edge from
  $b_{j}$ to $b_{j+1}$, and let
  \[
    \varphi(x) = \frac{\nu_{i} \cdot x}{h_{m}} + \frac{\mu_{j} \cdot x}{h_{l}}.
  \]
  At a vertex $a_{k}$ the second term vanishes and the first is at most $1$,
  with equality exactly for $k \in \{i, i+1\}$; at a vertex $b_{k}$ the first
  term vanishes and the second is at most $1$, with equality exactly for
  $k \in \{j, j+1\}$. A linear function attains its maximum over $P$ at
  vertices, so $\varphi \leq 1$ on $P$, and $\{\varphi = 1\}$ is a supporting
  hyperplane with
  \[
    P \cap \{\varphi = 1\} = \conv\{a_{i}, a_{i+1}, b_{j}, b_{j+1}\}.
  \]
  This face is $3$-dimensional: $a_{i}, a_{i+1}$ span the plane of
  $u_{1}, u_{2}$ and $b_{j}, b_{j+1}$ that of $u_{3}, u_{4}$, so the four
  points are linearly, hence affinely, independent. Thus each of the $ml$
  simplices is a facet, and since $P$ has exactly $ml$ facets, these are all
  of them; so $p$ realises $\mathcal{J}_{m,l}$.
\end{proof}

\begin{lemma}
  \label{lem:coherent}
  The orientation of $\mathcal{J}_{m,l}$ is coherent.
\end{lemma}

\begin{proof}
  Recall that deleting entry $a$ of a facet induces the orientation of the
  remaining triple if $a$ is even and the opposite one if $a$ is odd. A ridge
  consists of three of the four vertices of a facet
  $(a_{i}, a_{i+1}, b_{j}, b_{j+1})$, so it is of one of two kinds.
  A facet containing the ridge $\{a_{i}, b_{j}, b_{j+1}\}$ contains the edge
  $\{b_{j}, b_{j+1}\}$ of the second cycle and an edge of the first cycle
  through $a_{i}$, that is, $\{a_{i-1}, a_{i}\}$ or $\{a_{i}, a_{i+1}\}$. So
  this ridge lies in exactly the facets $(a_{i-1}, a_{i}, b_{j}, b_{j+1})$ and
  $(a_{i}, a_{i+1}, b_{j}, b_{j+1})$, which are different since $m \geq 3$.
  Deleting entry~$1$ of the first, an odd position, gives the opposite of
  $(a_{i}, b_{j}, b_{j+1})$; deleting entry~$2$ of the second, an even
  position, gives $(a_{i}, b_{j}, b_{j+1})$ itself.
  In the same way the ridge $\{a_{i}, a_{i+1}, b_{j}\}$ lies in exactly
  $(a_{i}, a_{i+1}, b_{j-1}, b_{j})$ and $(a_{i}, a_{i+1}, b_{j}, b_{j+1})$,
  which are different since $l \geq 3$. Deleting entry~$3$ of the first gives
  the opposite of $(a_{i}, a_{i+1}, b_{j})$, and deleting entry~$4$ of the
  second gives $(a_{i}, a_{i+1}, b_{j})$ itself.
  In both cases the ridge lies in exactly two facets, which induce opposite
  orientations on it.
\end{proof}

\begin{lemma}
  \label{lem:factor}
  For every $p \in (\mathbb{R}^{4})^{m+l}$, with $D(p)$ as in \eqref{eq:Djoin}
  and $A$, $B$ as in Definition~\ref{def:join},
  $D(p) = \langle A, \star B\rangle$. Hence $|D(p)| \leq |A|\,|B|$, and if
  $A, B \neq 0$, equality holds if and only if $\star B = \lambda A$ for some
  $\lambda \in \mathbb{R}$.
\end{lemma}

\begin{proof}
  Applying \eqref{eq:detstar} with $v = a_{i}$, $v' = a_{i+1}$, $w = b_{j}$
  and $w' = b_{j+1}$, each term of \eqref{eq:Djoin} is
  \[
    \det(a_{i}, a_{i+1}, b_{j}, b_{j+1})
    = \langle a_{i} \wedge a_{i+1}, \star(b_{j} \wedge b_{j+1})\rangle.
  \]
  The inner product is bilinear and $\star$ is linear, so summing over $i$
  and $j$ gives
  \[
    D(p) = \Bigl\langle \sum_{i \in \mathbb{Z}_{m}} a_{i} \wedge a_{i+1},\
      \star \sum_{j \in \mathbb{Z}_{l}} b_{j} \wedge b_{j+1} \Bigr\rangle
    = \langle A, \star B\rangle.
  \]
  By the Cauchy--Schwarz inequality, $|\langle A, \star B\rangle| \leq
  |A|\,|\star B|$, and $|\star B| = |B|$ since $\star$ is an isometry; hence
  $|D(p)| \leq |A|\,|B|$. If $A, B \neq 0$, then also $\star B \neq 0$, and
  equality in the Cauchy--Schwarz inequality for two nonzero vectors holds
  if and only if they are linearly dependent, that is, $\star B = \lambda A$
  for some $\lambda \in \mathbb{R}$.
\end{proof}
